    id SERIAL PRIMARY KEY,
    picture_large TEXT,
    name_title VARCHAR(50),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    gender VARCHAR(10),
    phone VARCHAR(20),
    email VARCHAR(100) UNIQUE NOT NULL,
    state VARCHAR(100),
    city VARCHAR(100),
    postcode INT
);
